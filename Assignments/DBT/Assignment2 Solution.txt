15. Write a query to count the total number of employees in each department.

select deptName, count(empId) as totalEmp
from department d
left join employees e
on d.deptId = e.deptId
group by d.deptId, d.deptName;


16. Write a query to list departments that have no employees. & create a view for those salespeople who belong to the city of indore. 


select * from department d
where not exists ( 
select *                                
from employees e
where e.deptId = d.deptId
);


create view indore_salesPerson as
select e.empName, e.location
from department d 
join employees e
on d.deptId = e.deptId
where d.deptName = 'Sales'
and e.location = 'Indore';


select * from indore_salesPerson;
